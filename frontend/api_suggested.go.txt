//  Ендпойнти:
//    GET /api/latest              -> { temp, humid, time }

import (
	"encoding/json"
	"math"
	"net/http"
	"strconv"
	"time"
)

// ---------- DTO ----------

type Reading struct {
	Temp  float64   `json:"temp"`
	Humid float64   `json:"humid"`
	Time  time.Time `json:"time"`
}

type Bucket struct {
	Label string    `json:"label"` // "01:00" для годин, "17.10" для днів
	Start time.Time `json:"start"` // початок інтервалу
	Temp  float64   `json:"temp"`  // середнє
	Humid float64   `json:"humid"` // середнє
	Count int       `json:"count"` // кількість вимірів в інтервалі
}

// ---------- допоміжне ----------

func writeJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	w.Header().Set("Access-Control-Allow-Origin", "*")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(v)
}

func queryInt(r *http.Request, key string, def int) int {
	if s := r.URL.Query().Get(key); s != "" {
		if n, err := strconv.Atoi(s); err == nil {
			return n
		}
	}
	return def
}

func clampInt(v, lo, hi int) int {
	if v < lo {
		return lo
	}
	if v > hi {
		return hi
	}
	return v
}

func round(v float64, digits int) float64 {
	f := math.Pow(10, float64(digits))
	return math.Round(v*f) / f
}

// ---------- читання з БД ----------

func queryReadings(since time.Time) ([]Reading, error) {
	rows, err := db.Query(
		"SELECT temperature, CAST(humidity AS REAL), time "+
			"FROM sensor_data WHERE time >= ? ORDER BY time ASC",
		since,
	)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var out []Reading
	for rows.Next() {
		var rd Reading
		if err := rows.Scan(&rd.Temp, &rd.Humid, &rd.Time); err != nil {
			return nil, err
		}
		out = append(out, rd)
	}
	return out, rows.Err()
}

// групування у кошики + середні
func aggregate(readings []Reading, trunc func(time.Time) time.Time, layout string) []Bucket {
	type acc struct {
		temp, humid float64
		n           int
	}

	byKey := map[time.Time]*acc{}
	order := []time.Time{}

	for _, rd := range readings {
		k := trunc(rd.Time)
		a, ok := byKey[k]
		if !ok {
			a = &acc{}
			byKey[k] = a
			order = append(order, k)
		}
		a.temp += rd.Temp
		a.humid += rd.Humid
		a.n++
	}

	out := make([]Bucket, 0, len(order))
	for _, k := range order {
		a := byKey[k]
		if a.n == 0 {
			continue
		}
		out = append(out, Bucket{
			Label: k.Format(layout),
			Start: k,
			Temp:  round(a.temp/float64(a.n), 2),
			Humid: round(a.humid/float64(a.n), 2),
			Count: a.n,
		})
	}
	return out
}

// ---------- handlers ----------

// GET /api/latest
func handleLatest(w http.ResponseWriter, r *http.Request) {
	var rd Reading
	err := db.QueryRow(
		"SELECT temperature, CAST(humidity AS REAL), time "+
			"FROM sensor_data ORDER BY time DESC LIMIT 1",
	).Scan(&rd.Temp, &rd.Humid, &rd.Time)

	if err != nil { // таблиця порожня або ще не створена
		writeJSON(w, http.StatusOK, nil)
		return
	}
	writeJSON(w, http.StatusOK, rd)
}

// GET /api/hourly?hours=24
func handleHourly(w http.ResponseWriter, r *http.Request) {
	hours := clampInt(queryInt(r, "hours", 24), 1, 24*30)

	readings, err := queryReadings(time.Now().Add(-time.Duration(hours) * time.Hour))
	if err != nil {
		writeJSON(w, http.StatusInternalServerError, map[string]string{"error": err.Error()})
		return
	}

	buckets := aggregate(readings, func(t time.Time) time.Time {
		return time.Date(t.Year(), t.Month(), t.Day(), t.Hour(), 0, 0, 0, t.Location())
	}, "15:04")

	writeJSON(w, http.StatusOK, buckets)
}

// GET /api/daily?days=7
func handleDaily(w http.ResponseWriter, r *http.Request) {
	days := clampInt(queryInt(r, "days", 7), 1, 365)

	from := time.Now().AddDate(0, 0, -(days - 1))
	since := time.Date(from.Year(), from.Month(), from.Day(), 0, 0, 0, 0, from.Location())

	readings, err := queryReadings(since)
	if err != nil {
		writeJSON(w, http.StatusInternalServerError, map[string]string{"error": err.Error()})
		return
	}

	buckets := aggregate(readings, func(t time.Time) time.Time {
		return time.Date(t.Year(), t.Month(), t.Day(), 0, 0, 0, 0, t.Location())
	}, "02.01") // -> "17.10"

	writeJSON(w, http.StatusOK, buckets)
}
